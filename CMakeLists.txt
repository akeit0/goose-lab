cmake_minimum_required(VERSION 3.27)
project(goose-lab LANGUAGES C CXX)
# Older dependency projects reset policies through their own minimum version.
# Keep our directory-scoped options effective without forcing consumer caches.
set(CMAKE_POLICY_DEFAULT_CMP0077 NEW)
set(CMAKE_POLICY_DEFAULT_CMP0126 NEW)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded")
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(GOOSE_LAB_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
option(GOOSE_LAB_GUI "Build text, Clay layout and GPU UI libraries" ON)
option(GOOSE_LAB_BUILD_SAMPLES "Build example applications" ON)
option(GOOSE_LAB_BUILD_TESTS "Build and register contract tests" ON)
option(GOOSE_LAB_BUILD_TOOLS "Build manual profiling tools" OFF)
find_package(Threads REQUIRED)
include(cmake/GooseToolchain.cmake)
include(cmake/NativeLibraries.cmake)
include(cmake/GoosePrograms.cmake)
if(GOOSE_LAB_BUILD_SAMPLES)
  include(cmake/Samples.cmake)
endif()
if(GOOSE_LAB_BUILD_TESTS)
  enable_testing()
  include(cmake/Contracts.cmake)
endif()
if(GOOSE_LAB_BUILD_TOOLS)
  include(cmake/Tools.cmake)
endif()
